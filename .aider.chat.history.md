"""Company Profile contract v1.

Typed business input only. Verified sections live under ``facts``.
``hypotheses`` and ``unknowns`` are separate collections and are not verified
business truth. This module is not used by workflows, APIs, or product data.
"""

from typing import Literal

from pydantic import BaseModel, ConfigDict, Field


class _StrictModel(BaseModel):
    model_config = ConfigDict(extra='forbid')


class Identity(_StrictModel):
    brand_name: str = Field(min_length=1)
    legal_name: str | None = None
    website: str | None = None
    category: str | None = None
    summary: str | None = None


class Offer(_StrictModel):
    summary: str | None = None
    products_or_services: list[str] = Field(default_factory=list)
    delivery_model: str | None = None
    geography: list[str] = Field(default_factory=list)


class Customers(_StrictModel):
    segments: list[str] = Field(default_factory=list)
    buyer_roles: list[str] = Field(default_factory=list)
    jobs_to_be_done: list[str] = Field(default_factory=list)
    exclusions: list[str] = Field(default_factory=list)


class CommercialOffer(_StrictModel):
    name: str = Field(min_length=1)
    price: str | None = None
    currency: str | None = None
    cadence: str | None = None
    notes: str | None = None


class Commercial(_StrictModel):
    summary: str | None = None
    currency: str | None = None
    offers: list[CommercialOffer] = Field(default_factory=list)
    notes: str | None = None


class ProofItem(_StrictModel):
    statement: str = Field(min_length=1)
    source: str | None = None


class Proof(_StrictModel):
    verified_claims: list[ProofItem] = Field(default_factory=list)
    assets: list[str] = Field(default_factory=list)


class Brand(_StrictModel):
    voice: str | None = None
    tone: list[str] = Field(default_factory=list)
    words_to_use: list[str] = Field(default_factory=list)
    words_to_avoid: list[str] = Field(default_factory=list)


class Channels(_StrictModel):
    owned: list[str] = Field(default_factory=list)
    active: list[str] = Field(default_factory=list)
    disallowed: list[str] = Field(default_factory=list)


class Outreach(_StrictModel):
    allowed_motions: list[str] = Field(default_factory=list)
    disallowed_motions: list[str] = Field(default_factory=list)
    requires_approval: bool = True
    notes: str | None = None


class Measurement(_StrictModel):
    north_star: str | None = None
    funnel: list[str] = Field(default_factory=list)
    known_baselines: list[str] = Field(default_factory=list)
    attribution_notes: str | None = None


class Constraints(_StrictModel):
    must_not_claim: list[str] = Field(default_factory=list)
    compliance_notes: list[str] = Field(default_factory=list)
    operational_limits: list[str] = Field(default_factory=list)


class VerifiedFacts(_StrictModel):
    """Verified business truth only. Hypotheses and unknowns are not valid here."""

    record_kind: Literal['verified'] = 'verified'
    identity: Identity
    offer: Offer
    customers: Customers
    commercial: Commercial
    proof: Proof
    brand: Brand
    channels: Channels
    outreach: Outreach
    measurement: Measurement
    constraints: Constraints


class Hypothesis(_StrictModel):
    record_kind: Literal['hypothesis'] = 'hypothesis'
    statement: str = Field(min_length=1)
    rationale: str | None = None


class Unknown(_StrictModel):
    record_kind: Literal['unknown'] = 'unknown'
    question: str = Field(min_length=1)
    why_it_matters: str | None = None


class CompanyProfileV1(_StrictModel):
    schema_version: Literal['1.0']
    facts: VerifiedFacts
    hypotheses: list[Hypothesis]
    unknowns: list[Unknown]
